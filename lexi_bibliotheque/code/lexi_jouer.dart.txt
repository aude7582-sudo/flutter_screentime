String _lxSans(String s) {
  const a = 'àâäáãåçéèêëíìîïñóòôöõúùûüýÿœæ';
  const b = 'aaaaaaceeeeiiiinooooouuuuyyoa';
  final l = s.toLowerCase();
  final o = StringBuffer();
  for (final ch in l.split('')) {
    final i = a.indexOf(ch);
    o.write(i >= 0 ? b[i] : ch);
  }
  return o.toString().replaceAll(RegExp(r"[^a-z0-9一-鿿]"), '');
}

List<Map<String, String>> _lxTirer(List<Map<String, String>> mots, int n,
    {bool Function(Map<String, String>)? filtre}) {
  final l = [...mots]..shuffle(_lxR);
  final res = <Map<String, String>>[];
  final ts = <String>{};
  final ds = <String>{};
  for (final m in l) {
    if (filtre != null && !filtre(m)) continue;
    if (ts.contains(m['t']) || ds.contains(m['d'])) continue;
    ts.add(m['t']!);
    ds.add(m['d']!);
    res.add(m);
    if (res.length >= n) break;
  }
  return res;
}

  int _entier(dynamic v) => v is num ? v.toInt() : 0;

  Future<void> _charger() async {
    try {
      final db = FirebaseFirestore.instance;
      final q = await db
          .collection('user_stats')
          .where('uid', isEqualTo: currentUserUid)
          .limit(1)
          .get();
      if (q.docs.isNotEmpty) {
        _ref = q.docs.first.reference;
        final d = q.docs.first.data();
        _langue = (d['langue'] ?? 'en').toString();
        final p = d['jeuxNiv'];
        if (p is Map) {
          final m = <String, Map<String, int>>{};
          p.forEach((k, v) {
            if (v is Map) {
              m[k.toString()] = {
                for (final e in v.entries) e.key.toString(): _entier(e.value)
              };
            }
          });
          _prog = m;
        }
      }
      await lexiParcours(_langue);
      final s = await db
          .collection('parcours')
          .where('langue', isEqualTo: _langue)
          .limit(450)
          .get();
      final l = <Map<String, String>>[];
      _ordre.clear();
      final docs = [...s.docs]
        ..sort((a, b) => _entier(a.data()['orderIndex'])
            .compareTo(_entier(b.data()['orderIndex'])));
      for (final doc in docs) {
        final m = doc.data();
        final t = (m['term'] ?? '').toString().trim();
        final d = (m['definition'] ?? '').toString().trim();
        if (t.isNotEmpty && d.isNotEmpty) {
          l.add({'t': t, 'd': d});
          _ordre.add(_entier(m['orderIndex']));
        }
      }
      _tous = l;
    } catch (_) {}
    if (mounted) setState(() => _charge = true);
  }

  /// Plus le niveau monte, plus on prend de mots (et des mots plus durs).
  List<Map<String, String>> _pool(int niveau) {
    final lim = 20 + 10 * niveau;
    final l = <Map<String, String>>[];
    for (var i = 0; i < _tous.length; i++) {
      if (_ordre[i] < lim) l.add(_tous[i]);
    }
    return l.length >= 14 ? l : _tous;
  }

  Future<void> _dire(String texte) async {
    try {
      await _tts.stop();
      await _tts.setLanguage(_lxVoix[_langue] ?? 'en-US');
      await _tts.setSpeechRate(0.42);
      await _tts.speak(texte);
    } catch (_) {}
  }

  int _etoiles(String jeu, int n) => _prog[jeu]?[n.toString()] ?? 0;

  bool _ouvert(String jeu, int n) => n == 1 || _etoiles(jeu, n - 1) > 0;

  int _total(String jeu) =>
      (_prog[jeu]?.values ?? const <int>[]).fold(0, (a, b) => a + b);

  int _niveauActuel(String jeu) {
    var n = 1;
    while (n < 10 && _etoiles(jeu, n) > 0) {
      n++;
    }
    return n;
  }

  Future<void> _gagne(String jeu, int niveau, int etoiles, int score) async {
    final avant = _etoiles(jeu, niveau);
    if (etoiles > avant) {
      setState(() {
        _prog.putIfAbsent(jeu, () => {});
        _prog[jeu]![niveau.toString()] = etoiles;
      });
    }
    try {
      final maj = <String, dynamic>{
        'totalXp': FieldValue.increment(score),
        'jeuxTotal': FieldValue.increment(1),
      };
      if (etoiles > avant) {
        maj['jeuxNiv.' + jeu + '.' + niveau.toString()] = etoiles;
      }
      if (score > 0 || etoiles > avant) await _ref?.update(maj);
    } catch (_) {}
  }

      _niveau = null;
    });
  }

// 1. Paires (manches avec « Bravo ! » entre chaque)

class _JeuPaires extends StatefulWidget {
  const _JeuPaires({required this.a});
  final _LxArgs a;
  @override
  State<_JeuPaires> createState() => _JeuPairesState();
}

class _JeuPairesState extends State<_JeuPaires> {
  late int _manches;
  late int _n;
  int _manche = 0;
  int _erreurs = 0;
  int _score = 0;
  bool _pause = false;
  List<Map<String, String>> _g = [];
  List<Map<String, String>> _d = [];
  String? _selG;
  String? _selD;
  String? _rouge;
  final Set<String> _trouves = {};

  int get _maxErreurs => 3 + 2 * _manches;

  @override
  void initState() {
    super.initState();
    final n = widget.a.niveau;
    _manches = n <= 3 ? 2 : (n <= 7 ? 3 : 4);
    _n = math.min(4 + (n - 1) ~/ 3, 7);
    _nouvelle();
  }

  void _nouvelle() {
    final l = _lxTirer(widget.a.mots, _n);
    _g = [...l]..shuffle(_lxR);
    _d = [...l]..shuffle(_lxR);
    _trouves.clear();
    _selG = null;
    _selD = null;
  }

  void _verifier() {
    if (_selG == null || _selD == null) return;
    final m = _g.firstWhere((x) => x['t'] == _selG);
    if (m['d'] == _selD) {
      _trouves.add(_selG!);
      _score += 2;
      widget.a.dire(_selG!);
      _selG = null;
      _selD = null;
      if (_trouves.length == _g.length) {
        if (_manche + 1 >= _manches) {
          final e = _erreurs <= 1 ? 3 : (_erreurs <= 4 ? 2 : 1);
          Future.delayed(const Duration(milliseconds: 400),
              () => widget.a.fin(e, _score + 5 * e));
        } else {
          Future.delayed(const Duration(milliseconds: 350), () {
            if (mounted) setState(() => _pause = true);
          });
        }
      }
    } else {
      _erreurs++;
      _rouge = _selG! + '|' + _selD!;
      _selG = null;
      _selD = null;
      Future.delayed(const Duration(milliseconds: 600), () {
        if (mounted) setState(() => _rouge = null);
      });
      if (_erreurs >= _maxErreurs) {
        Future.delayed(const Duration(milliseconds: 500),
            () => widget.a.fin(0, _score));
      }
    }
  }

  Widget _tuile(_LxP p, String texte, bool gauche) {
    final cle = gauche ? texte : null;
    final trouve = gauche
        ? _trouves.contains(texte)
        : _g.any((m) => m['d'] == texte && _trouves.contains(m['t']));
    final sel = gauche ? _selG == texte : _selD == texte;
    final rouge = _rouge != null &&
        (gauche ? _rouge!.split('|')[0] == texte : _rouge!.split('|')[1] == texte);
    Color fond = p.carte;
    Color bord = p.bord;
    if (trouve) {
      fond = _lxVert.withOpacity(0.15);
      bord = _lxVert;
    } else if (rouge) {
      fond = _lxRouge.withOpacity(0.15);
      bord = _lxRouge;
    } else if (sel) {
      fond = widget.a.couleur.withOpacity(0.15);
      bord = widget.a.couleur;
    }
    return Padding(
      padding: const EdgeInsets.only(bottom: 10),
      child: GestureDetector(
        onTap: trouve
            ? null
            : () => setState(() {
                  if (gauche) {
                    _selG = cle;
                  } else {
                    _selD = texte;
                  }
                  _verifier();
                }),
        child: AnimatedContainer(
          duration: const Duration(milliseconds: 150),
          height: 58,
          alignment: Alignment.center,
          padding: const EdgeInsets.symmetric(horizontal: 8),
          decoration: BoxDecoration(
            color: fond,
            borderRadius: BorderRadius.circular(14),
            border: Border.all(color: bord, width: 2),
          ),
          child: Opacity(
            opacity: trouve ? 0.45 : 1,
            child: Text(texte,
                textAlign: TextAlign.center,
                maxLines: 2,
                overflow: TextOverflow.ellipsis,
                style: TextStyle(
                    color: p.texte,
                    fontSize: 15,
                    fontWeight: FontWeight.w700)),
          ),
        ),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final p = _lxP(context);
    if (_pause) {
      return _lxEntreManches(p, _manche + 1, _manches, () {
        setState(() {
          _pause = false;
          _manche++;
          _nouvelle();
        });
      });
    }
    return ListView(
      padding: const EdgeInsets.fromLTRB(18, 10, 18, 20),
      children: [
        Row(children: [
          Text('Manche ' + (_manche + 1).toString() + '/' + _manches.toString(),
              style: TextStyle(
                  color: p.texte, fontSize: 15, fontWeight: FontWeight.w800)),
          const Spacer(),
          Text('Erreurs : ' + _erreurs.toString() + '/' + _maxErreurs.toString(),
              style: TextStyle(color: p.gris, fontSize: 13.5)),
        ]),
        const SizedBox(height: 14),
        Row(crossAxisAlignment: CrossAxisAlignment.start, children: [
          Expanded(
              child: Column(children: [
            for (final m in _g) _tuile(p, m['t']!, true),
          ])),
          const SizedBox(width: 12),
          Expanded(
              child: Column(children: [
            for (final m in _d) _tuile(p, m['d']!, false),
          ])),
        ]),
      ],
    );
  }
}

// 2. Mots mêlés

class _JeuMeles extends StatefulWidget {
  const _JeuMeles({required this.a});
  final _LxArgs a;
  @override
  State<_JeuMeles> createState() => _JeuMelesState();
}

class _JeuMelesState extends State<_JeuMeles> {
  late int _n;
  List<String> _grille = [];
  final List<Map<String, dynamic>> _mots = [];
  int? _debut;
  final Set<int> _cases = {};
  final Stopwatch _chrono = Stopwatch()..start();
  int _score = 0;

  @override
  void initState() {
    super.initState();
    _generer();
  }

  String _propre(String t) {
    if (widget.a.zh) return t.replaceAll(' ', '');
    return _lxSans(t).toUpperCase();
  }

  void _generer() {
    final niv = widget.a.niveau;
    _n = widget.a.zh ? 6 : math.min(6 + (niv - 1) ~/ 3, 9);
    final nb = math.min(3 + (niv - 1) ~/ 2, 7);
    final dirs = <List<int>>[
      [0, 1],
      [1, 0]
    ];
    if (niv >= 4) dirs.add([1, 1]);
    if (niv >= 7) {
      dirs.addAll([
        [0, -1],
        [-1, 0],
        [1, -1],
        [-1, -1]
      ]);
    }
    final g = List<String>.filled(_n * _n, '');
    final candidats = _lxTirer(widget.a.mots, 40, filtre: (m) {
      final w = _propre(m['t']!);
      final min = widget.a.zh ? 2 : 3;
      if (w.length < min || w.length > _n) return false;
      if (!widget.a.zh && !RegExp(r'^[A-Z]+$').hasMatch(w)) return false;
      return true;
    });
    _mots.clear();
    for (final m in candidats) {
      if (_mots.length >= nb) break;
      final w = _propre(m['t']!);
      var place = false;
      for (var essai = 0; essai < 200 && !place; essai++) {
        final dir = dirs[_lxR.nextInt(dirs.length)];
        final r = _lxR.nextInt(_n);
        final c = _lxR.nextInt(_n);
        final rf = r + dir[0] * (w.length - 1);
        final cf = c + dir[1] * (w.length - 1);
        if (rf < 0 || rf >= _n || cf < 0 || cf >= _n) continue;
        var ok = true;
        for (var k = 0; k < w.length; k++) {
          final i = (r + dir[0] * k) * _n + (c + dir[1] * k);
          if (g[i] != '' && g[i] != w[k]) {
            ok = false;
            break;
          }
        }
        if (!ok) continue;
        final idx = <int>[];
        for (var k = 0; k < w.length; k++) {
          final i = (r + dir[0] * k) * _n + (c + dir[1] * k);
          g[i] = w[k];
          idx.add(i);
        }
        _mots.add({'w': w, 't': m['t'], 'd': m['d'], 'idx': idx, 'ok': false});
        place = true;
      }
    }
    String lettres;
    if (widget.a.zh) {
      lettres = _mots.map((m) => m['w'] as String).join() + '的一是了我不人在他有这个上们来到时大地为子中你说生国年着就那和要';
    } else {
      lettres = 'ABCDEFGHIJKLMNOPRSTUVWY' + _mots.map((m) => m['w']).join();
    }
    for (var i = 0; i < g.length; i++) {
      if (g[i] == '') g[i] = lettres[_lxR.nextInt(lettres.length)];
    }
    _grille = g;
  }

  void _toucher(int i) {
    if (_debut == null) {
      setState(() => _debut = i);
      return;
    }
    final a = _debut!;
    setState(() => _debut = null);
    final r1 = a ~/ _n, c1 = a % _n, r2 = i ~/ _n, c2 = i % _n;
    final dr = (r2 - r1).sign, dc = (c2 - c1).sign;
    final lg = math.max((r2 - r1).abs(), (c2 - c1).abs()) + 1;
    if (!(r1 == r2 || c1 == c2 || (r2 - r1).abs() == (c2 - c1).abs())) return;
    final idx = <int>[];
    for (var k = 0; k < lg; k++) {
      idx.add((r1 + dr * k) * _n + (c1 + dc * k));
    }
    final mot = idx.map((x) => _grille[x]).join();
    final inv = mot.split('').reversed.join();
    for (final m in _mots) {
      if (m['ok'] == true) continue;
      if (m['w'] == mot || m['w'] == inv) {
        setState(() {
          m['ok'] = true;
          _cases.addAll(idx);
          _score += 3;
        });
        widget.a.dire(m['t'] as String);
        if (_mots.every((x) => x['ok'] == true)) {
          final s = _chrono.elapsed.inSeconds;
          final par = 25 * _mots.length;
          final e = s <= par ? 3 : (s <= par * 2 ? 2 : 1);
          Future.delayed(const Duration(milliseconds: 500),
              () => widget.a.fin(e, _score + 5 * e));
        }
        return;
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    final p = _lxP(context);
    final cacher = widget.a.niveau >= 6;
    return ListView(
      padding: const EdgeInsets.fromLTRB(16, 8, 16, 20),
      children: [
        Text(
            cacher
                ? 'Trouve en ' +
                    (widget.a.langue == 'en' ? 'anglais' : 'langue étrangère') +
                    ' les mots de cette liste :'
                : 'Touche la 1re lettre, puis la dernière :',
            style: TextStyle(color: p.gris, fontSize: 13.5)),
        const SizedBox(height: 8),
        Wrap(spacing: 8, runSpacing: 8, children: [
          for (final m in _mots)
            Container(
              padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 6),
              decoration: BoxDecoration(
                color: m['ok'] == true ? _lxVert.withOpacity(0.15) : p.carte,
                borderRadius: BorderRadius.circular(10),
                border: Border.all(
                    color: m['ok'] == true ? _lxVert : p.bord),
              ),
              child: Text(
                  m['ok'] == true
                      ? (m['t'] as String) + ' = ' + (m['d'] as String)
                      : (cacher ? m['d'] as String : m['t'] as String),
                  style: TextStyle(
                      color: p.texte,
                      fontSize: 13.5,
                      fontWeight: FontWeight.w700,
                      decoration: m['ok'] == true
                          ? TextDecoration.lineThrough
                          : null)),
            ),
        ]),
        const SizedBox(height: 14),
        AspectRatio(
          aspectRatio: 1,
          child: GridView.builder(
            physics: const NeverScrollableScrollPhysics(),
            gridDelegate: SliverGridDelegateWithFixedCrossAxisCount(
                crossAxisCount: _n, mainAxisSpacing: 4, crossAxisSpacing: 4),
            itemCount: _grille.length,
            itemBuilder: (_, i) {
              final trouve = _cases.contains(i);
              final debut = _debut == i;
              return GestureDetector(
                onTap: () => _toucher(i),
                child: Container(
                  alignment: Alignment.center,
                  decoration: BoxDecoration(
                    color: debut
                        ? widget.a.couleur
                        : (trouve ? _lxVert.withOpacity(0.25) : p.carte),
                    borderRadius: BorderRadius.circular(8),
                    border: Border.all(color: p.bord),
                  ),
                  child: Text(_grille[i],
                      style: TextStyle(
                          color: debut ? Colors.white : p.texte,
                          fontSize: _n >= 9 ? 15 : 18,
                          fontWeight: FontWeight.w800)),
                ),
              );
            },
          ),
        ),
      ],
    );
  }
}

class _JeuMemoire extends StatefulWidget {
  const _JeuMemoire({required this.a});
  final _LxArgs a;
  @override
  State<_JeuMemoire> createState() => _JeuMemoireState();
}

class _JeuMemoireState extends State<_JeuMemoire> {
  final List<Map<String, String>> _cartes = [];
  final Set<int> _ouvertes = {};
  final Set<int> _trouvees = {};
  int _coups = 0;
  bool _bloque = false;
  late int _paires;

  @override
  void initState() {
    super.initState();
    _paires = math.min(4 + (widget.a.niveau - 1) ~/ 2, 9);
    for (final m in _lxTirer(widget.a.mots, _paires)) {
      _cartes.add({'id': m['t']!, 'x': m['t']!, 'lang': '1'});
      _cartes.add({'id': m['t']!, 'x': m['d']!, 'lang': '0'});
    }
    _cartes.shuffle(_lxR);
  }

  void _retourner(int i) {
    if (_bloque || _ouvertes.contains(i) || _trouvees.contains(i)) return;
    setState(() => _ouvertes.add(i));
    if (_cartes[i]['lang'] == '1') widget.a.dire(_cartes[i]['x']!);
    if (_ouvertes.length == 2) {
      _coups++;
      final a = _ouvertes.first, b = _ouvertes.last;
      if (_cartes[a]['id'] == _cartes[b]['id']) {
        setState(() {
          _trouvees.addAll([a, b]);
          _ouvertes.clear();
        });
        if (_trouvees.length == _cartes.length) {
          final e = _coups <= _paires * 1.5
              ? 3

            _ouvertes.clear();
            _bloque = false;
          });
        });
      }
    }

        padding: const EdgeInsets.fromLTRB(18, 8, 18, 8),

List<String> _lxMauvais(List<Map<String, String>> mots, String bon, String cle,
    int n) {
  final l = mots.map((m) => m[cle]!).where((x) => x != bon).toSet().toList()
    ..shuffle(_lxR);
  return l.take(n).toList();
}

// 4. Duel éclair (contre le renard)

class _JeuDuel extends StatefulWidget {
  const _JeuDuel({required this.a});
  final _LxArgs a;
  @override
  State<_JeuDuel> createState() => _JeuDuelState();
}

class _JeuDuelState extends State<_JeuDuel> {
  late int _cible;
  late double _total;
  double _reste = 0;
  int _moi = 0;
  double _renard = 0;
  Map<String, String>? _q;
  List<String> _choix = [];
  String? _clique;
  Timer? _t;

  @override
  void initState() {
    super.initState();
    final n = widget.a.niveau;
    _cible = 5 + n;
    _total = (42 - 2 * n).toDouble();
    _reste = _total;
    _suivante();
    _t = Timer.periodic(const Duration(milliseconds: 100), (_) {
      if (!mounted) return;
      setState(() {
        _reste -= 0.1;
        _renard += _cible / (_total * 0.92) * 0.1;
      });
      if (_renard >= _cible || _reste <= 0) {
        _t?.cancel();
        widget.a.fin(0, _moi);
      }
    });
  }

  @override
  void dispose() {
    _t?.cancel();
    super.dispose();
  }

  void _suivante() {
    final m = widget.a.mots[_lxR.nextInt(widget.a.mots.length)];
    _q = m;
    _choix = [m['d']!, ..._lxMauvais(widget.a.mots, m['d']!, 'd', 3)]
      ..shuffle(_lxR);
    _clique = null;
  }

  void _repondre(String c) {
    if (_clique != null) return;
    final bon = c == _q!['d'];
    setState(() => _clique = c);
    if (bon) {
      _moi++;
      widget.a.dire(_q!['t']!);
      if (_moi >= _cible) {
        _t?.cancel();
        final f = _reste / _total;
        final e = f > 0.35 ? 3 : (f > 0.15 ? 2 : 1);
        Future.delayed(const Duration(milliseconds: 300),
            () => widget.a.fin(e, _moi + 5 * e));
        return;
      }
    }
    Future.delayed(Duration(milliseconds: bon ? 250 : 700), () {
      if (mounted) setState(_suivante);
    });
  }

  Widget _piste(_LxP p, String e, double v, Color c) => Row(children: [
        Text(e, style: const TextStyle(fontSize: 22)),
        const SizedBox(width: 8),
        Expanded(
          child: ClipRRect(
            borderRadius: BorderRadius.circular(8),
            child: LinearProgressIndicator(
              value: (v / _cible).clamp(0.0, 1.0),
              minHeight: 12,
              backgroundColor: p.doux,
              valueColor: AlwaysStoppedAnimation(c),
            ),
          ),
        ),
        const SizedBox(width: 8),
        Text(v.floor().toString() + '/' + _cible.toString(),
            style: TextStyle(color: p.gris, fontWeight: FontWeight.w800)),
      ]);

  @override
  Widget build(BuildContext context) {
    final p = _lxP(context);
    final q = _q!;
    return ListView(
      padding: const EdgeInsets.fromLTRB(18, 10, 18, 20),
      children: [
        Row(children: [
          Icon(Icons.timer_rounded, color: _reste < 8 ? _lxRouge : p.gris),
          const SizedBox(width: 4),
          Text(math.max(0, _reste).ceil().toString() + ' s',
              style: TextStyle(
                  color: _reste < 8 ? _lxRouge : p.texte,
                  fontSize: 17,
                  fontWeight: FontWeight.w900)),
        ]),
        const SizedBox(height: 12),
        _piste(p, '🙋', _moi.toDouble(), _lxVert),
        const SizedBox(height: 8),
        _piste(p, '🦊', _renard, _lxOrange),
        const SizedBox(height: 24),
        Text(q['t']!,
            textAlign: TextAlign.center,
            style: TextStyle(
                color: p.texte, fontSize: 32, fontWeight: FontWeight.w900)),
        const SizedBox(height: 22),
        for (final c in _choix)
          _lxChoix(
              p,
              c,
              _clique == null
                  ? null
                  : (c == q['d'] ? _lxVert : (c == _clique ? _lxRouge : null)),
              () => _repondre(c)),
      ],
    );
  }
}

// 5. Lettres mélangées

class _JeuLettres extends StatefulWidget {
  const _JeuLettres({required this.a});
  final _LxArgs a;
  @override
  State<_JeuLettres> createState() => _JeuLettresState();
}

class _JeuLettresState extends State<_JeuLettres> {
  List<Map<String, String>> _liste = [];
  int _i = 0;
  List<String> _tuiles = [];
  final List<int> _choisies = [];
  int _erreurs = 0;
  int _score = 0;
  bool _rouge = false;
  bool _indice = false;

  String _mot(Map<String, String> m) =>
      widget.a.zh ? m['t']!.replaceAll(' ', '') : m['t']!.toUpperCase();

  @override
  void initState() {
    super.initState();
    final n = widget.a.niveau;
    final maxL = 4 + (n - 1) ~/ 2;
    _liste = _lxTirer(widget.a.mots, n >= 6 ? 8 : 6, filtre: (m) {
      final w = _mot(m);
      if (widget.a.zh) return w.length >= 2 && w.length <= 4;
      if (!RegExp(r"^[A-ZÀ-ÿ]+$").hasMatch(w)) return false;
      return w.length >= 3 && w.length <= maxL + 1;
    });
    if (_liste.length < 4) {
      _liste = _lxTirer(widget.a.mots, 6, filtre: (m) {
        final w = _mot(m);
        return widget.a.zh ? w.length >= 2 : (w.length >= 3 && w.length <= 9 && !w.contains(' '));
      });
    }
    _preparer();
  }

  void _preparer() {
    final w = _mot(_liste[_i]);
    final l = w.split('');
    do {
      l.shuffle(_lxR);
    } while (l.join() == w && w.length > 1 && l.toSet().length > 1);
    _tuiles = l;
    _choisies.clear();
    _indice = false;
  }

  void _toucher(int k) {
    if (_choisies.contains(k)) return;
    setState(() => _choisies.add(k));
    final w = _mot(_liste[_i]);
    if (_choisies.length == w.length) {
      final rep = _choisies.map((x) => _tuiles[x]).join();
      if (rep == w) {
        _score += 3;
        widget.a.dire(_liste[_i]['t']!);
        Future.delayed(const Duration(milliseconds: 500), () {
          if (!mounted) return;
          if (_i + 1 >= _liste.length) {
            final e = _erreurs == 0 ? 3 : (_erreurs <= 2 ? 2 : 1);
            widget.a.fin(e, _score + 5 * e);
          } else {
            setState(() {
              _i++;
              _preparer();
            });
          }
        });
      } else {
        _erreurs++;
        setState(() => _rouge = true);
        Future.delayed(const Duration(milliseconds: 600), () {
          if (!mounted) return;
          setState(() {
            _rouge = false;
            _choisies.clear();
          });
          if (_erreurs >= 5) widget.a.fin(0, _score);
        });
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    final p = _lxP(context);
    final m = _liste[_i];
    final w = _mot(m);
    return ListView(
      padding: const EdgeInsets.fromLTRB(18, 10, 18, 20),
      children: [
        Row(children: [
          Text('Mot ' + (_i + 1).toString() + '/' + _liste.length.toString(),
              style: TextStyle(
                  color: p.texte, fontSize: 15, fontWeight: FontWeight.w800)),
          const Spacer(),
          Text('Erreurs : ' + _erreurs.toString() + '/5',
              style: TextStyle(color: p.gris, fontSize: 13.5)),
        ]),
        const SizedBox(height: 20),
        Text('🇫🇷 ' + m['d']!,
            textAlign: TextAlign.center,
            style: TextStyle(
                color: p.texte, fontSize: 22, fontWeight: FontWeight.w800)),
        if (_indice)
          Padding(
            padding: const EdgeInsets.only(top: 6),
            child: Text('Commence par « ' + w[0] + ' »',
                textAlign: TextAlign.center,
                style: const TextStyle(color: _lxOrange, fontSize: 14)),
          ),
        const SizedBox(height: 22),
        Wrap(alignment: WrapAlignment.center, spacing: 6, runSpacing: 6, children: [
          for (var k = 0; k < w.length; k++)
            Container(
              width: 38,
              height: 46,
              alignment: Alignment.center,
              decoration: BoxDecoration(
                color: _rouge ? _lxRouge.withOpacity(0.15) : p.carte,
                borderRadius: BorderRadius.circular(10),
                border: Border.all(
                    color: _rouge
                        ? _lxRouge
                        : (k < _choisies.length ? widget.a.couleur : p.bord),
                    width: 2),
              ),
              child: Text(k < _choisies.length ? _tuiles[_choisies[k]] : '',
                  style: TextStyle(
                      color: p.texte,
                      fontSize: 20,
                      fontWeight: FontWeight.w900)),
            ),
        ]),
        const SizedBox(height: 26),
        Wrap(alignment: WrapAlignment.center, spacing: 10, runSpacing: 10, children: [
          for (var k = 0; k < _tuiles.length; k++)
            GestureDetector(
              onTap: () => _toucher(k),
              child: AnimatedOpacity(
                duration: const Duration(milliseconds: 150),
                opacity: _choisies.contains(k) ? 0.2 : 1,
                child: Container(
                  width: 50,
                  height: 54,
                  alignment: Alignment.center,
                  decoration: BoxDecoration(
                    color: widget.a.couleur,
                    borderRadius: BorderRadius.circular(12),
                  ),
                  child: Text(_tuiles[k],
                      style: const TextStyle(
                          color: Colors.white,
                          fontSize: 22,
                          fontWeight: FontWeight.w900)),
                ),
              ),
            ),
        ]),
        const SizedBox(height: 24),
        Row(children: [
          Expanded(
            child: OutlinedButton.icon(
              onPressed: _choisies.isEmpty
                  ? null
                  : () => setState(() => _choisies.removeLast()),
              icon: const Icon(Icons.backspace_outlined),
              label: const Text('Effacer'),
              style: OutlinedButton.styleFrom(
                  foregroundColor: p.texte,
                  side: BorderSide(color: p.bord),
                  padding: const EdgeInsets.symmetric(vertical: 13)),
            ),
          ),
          if (widget.a.niveau <= 5) ...[
            const SizedBox(width: 10),
            Expanded(
              child: OutlinedButton.icon(
                onPressed: _indice ? null : () => setState(() => _indice = true),
                icon: const Icon(Icons.lightbulb_outline_rounded),
                label: const Text('Indice'),
                style: OutlinedButton.styleFrom(
                    foregroundColor: _lxOrange,
                    side: const BorderSide(color: _lxOrange),
                    padding: const EdgeInsets.symmetric(vertical: 13)),
              ),
            ),
          ],
        ]),
      ],
    );
  }
}

  late double _duree;
  int _i = 0;
  int _vies = 3;
  int _score = 0;
  double _reste = 1;
  Map<String, String>? _q;
  String _prop = '';
  bool _vrai = true;
  Color? _flash;
  Timer? _t;
  bool _fini = false;

  @override
  void initState() {
    super.initState();
    final n = widget.a.niveau;
    _total = 10 + n;
    _duree = math.max(2.2, 6.2 - 0.4 * n);
    _suivante();
    _t = Timer.periodic(const Duration(milliseconds: 50), (_) {
      if (!mounted || _flash != null || _fini) return;
      setState(() => _reste -= 0.05 / _duree);
      if (_reste <= 0) _repondre(null);
    });
  }

  @override
  void dispose() {
    _t?.cancel();
    super.dispose();
  }

    final m = widget.a.mots[_lxR.nextInt(widget.a.mots.length)];

    _vrai = _lxR.nextBool();

    _reste = 1;

    if (_flash != null || _fini) return;
    final bon = r != null && r == _vrai;
    if (bon) {
      _score += 2;
    } else {
      _vies--;
    }
    setState(() => _flash = bon ? _lxVert : _lxRouge);

      if (!mounted) return;
      if (_vies <= 0) {
        _fini = true;
        widget.a.fin(0, _score);
        return;
      }
      if (_i + 1 >= _total) {
        _fini = true;
        widget.a.fin(_vies, _score + 5 * _vies);
        return;
      }
      setState(() {
        _i++;
        _flash = null;
        _suivante();
      });
    });
  }

  @override
  Widget build(BuildContext context) {
    final p = _lxP(context);
    return Padding(
      padding: const EdgeInsets.fromLTRB(18, 10, 18, 20),

              style: TextStyle(
                  color: p.texte, fontSize: 15, fontWeight: FontWeight.w800)),
          const Spacer(),
          _lxVies(_vies, 3),
        ]),
        const SizedBox(height: 10),
        ClipRRect(
          borderRadius: BorderRadius.circular(8),
          child: LinearProgressIndicator(
            value: _reste.clamp(0.0, 1.0),
            minHeight: 10,
            backgroundColor: p.doux,
            valueColor: AlwaysStoppedAnimation(
                _reste < 0.3 ? _lxRouge : widget.a.couleur),

          padding: const EdgeInsets.symmetric(vertical: 36, horizontal: 16),
          decoration: BoxDecoration(
            color: _flash == null ? p.carte : _flash!.withOpacity(0.18),

            Text('=', style: TextStyle(color: p.gris, fontSize: 22)),

                  borderRadius: BorderRadius.circular(16),

            ),
            onSubmitted: _valider,
          ),
          const SizedBox(height: 16),

// 8. Pluie de mots

class _JeuPluie extends StatefulWidget {
  const _JeuPluie({required this.a});
  final _LxArgs a;
  @override
  State<_JeuPluie> createState() => _JeuPluieState();
}

class _JeuPluieState extends State<_JeuPluie> {
  late int _total;
  late double _duree;
  int _i = 0;
  int _vies = 3;
  int _score = 0;
  double _y = 0;
  double _x = 0.5;
  Map<String, String>? _q;
  List<String> _choix = [];
  Color? _flash;
  Timer? _t;
  bool _fini = false;

  @override
  void initState() {
    super.initState();
    final n = widget.a.niveau;
    _total = 8 + n;
    _duree = math.max(2.4, 7.5 - 0.5 * n);
    _suivant();
    _t = Timer.periodic(const Duration(milliseconds: 30), (_) {
      if (!mounted || _flash != null || _fini) return;
      setState(() => _y += 0.03 / _duree);
      if (_y >= 1) _repondre(null);
    });
  }

  @override
  void dispose() {
    _t?.cancel();
    super.dispose();
  }

  void _suivant() {
    final m = widget.a.mots[_lxR.nextInt(widget.a.mots.length)];
    _q = m;
    final nb = widget.a.niveau >= 6 ? 3 : 2;
    _choix = [m['d']!, ..._lxMauvais(widget.a.mots, m['d']!, 'd', nb)]
      ..shuffle(_lxR);
    _y = 0;
    _x = 0.15 + _lxR.nextDouble() * 0.7;
  }

  void _repondre(String? c) {
    if (_flash != null || _fini) return;
    final bon = c != null && c == _q!['d'];
    if (bon) {
      _score += 2;
      widget.a.dire(_q!['t']!);
    } else {
      _vies--;
    }
    setState(() => _flash = bon ? _lxVert : _lxRouge);
    Future.delayed(const Duration(milliseconds: 500), () {
      if (!mounted) return;
      if (_vies <= 0) {
        _fini = true;
        widget.a.fin(0, _score);
        return;
      }
      if (_i + 1 >= _total) {
        _fini = true;
        widget.a.fin(_vies, _score + 5 * _vies);
        return;
      }
      setState(() {
        _i++;
        _flash = null;
        _suivant();
      });
    });
  }

  @override
  Widget build(BuildContext context) {
    final p = _lxP(context);
    return Column(children: [
      Padding(
        padding: const EdgeInsets.fromLTRB(18, 6, 18, 6),
        child: Row(children: [
          Text((_i + 1).toString() + '/' + _total.toString(),
              style: TextStyle(
                  color: p.texte, fontSize: 15, fontWeight: FontWeight.w800)),
          const Spacer(),
          _lxVies(_vies, 3),
        ]),
      ),
      Expanded(
        child: LayoutBuilder(builder: (_, c) {
          final h = c.maxHeight;
          final w = c.maxWidth;
          return Container(
            margin: const EdgeInsets.symmetric(horizontal: 14),
            decoration: BoxDecoration(
              gradient: LinearGradient(
                  begin: Alignment.topCenter,
                  end: Alignment.bottomCenter,
                  colors: [widget.a.couleur.withOpacity(0.18), p.fond]),
              borderRadius: BorderRadius.circular(22),
            ),
            child: Stack(children: [
              Positioned(
                left: 0,
                right: 0,
                bottom: 0,
                child: Container(
                    height: 6,
                    decoration: BoxDecoration(
                        color: _lxRouge.withOpacity(0.5),
                        borderRadius: BorderRadius.circular(3))),
              ),
              Positioned(
                top: _y.clamp(0.0, 1.0) * (h - 60),
                left: (_x * (w - 28 - 150)).clamp(0.0, w),
                child: Container(
                  width: 150,
                  padding:
                      const EdgeInsets.symmetric(horizontal: 10, vertical: 12),
                  decoration: BoxDecoration(
                    color: _flash ?? widget.a.couleur,
                    borderRadius: BorderRadius.circular(18),
                  ),
                  child: Text('💧 ' + _q!['t']!,
                      textAlign: TextAlign.center,
                      maxLines: 2,
                      overflow: TextOverflow.ellipsis,
                      style: const TextStyle(
                          color: Colors.white,
                          fontSize: 17,
                          fontWeight: FontWeight.w900)),
                ),
              ),
            ]),
          );
        }),
      ),
      Padding(
        padding: const EdgeInsets.fromLTRB(18, 12, 18, 12),
        child: Column(children: [
          for (final c in _choix)
            _lxChoix(
                p,
                c,
                _flash == null ? null : (c == _q!['d'] ? _lxVert : null),
                () => _repondre(c)),
        ]),
      ),
    ]);
  }
}

// 9. Le mot mystère (comme Wordle)

class _JeuMotMystere extends StatefulWidget {
  const _JeuMotMystere({required this.a});
  final _LxArgs a;
  @override
  State<_JeuMotMystere> createState() => _JeuMotMystereState();
}

class _JeuMotMystereState extends State<_JeuMotMystere> {
  String _mot = '';
  String _trad = '';
  final List<String> _essais = [];
  String _saisie = '';
  bool _fini = false;

  @override
  void initState() {
    super.initState();
    final n = widget.a.niveau;
    final lg = widget.a.langue;
    final Map<String, List<String>> src =
        n <= 3 ? _lxMots4 : (n <= 7 ? _lxMots5 : _lxMots6);
    final liste = src[lg];
    if (liste != null) {
      final idx = n <= 3 ? n - 1 : (n <= 7 ? n - 4 : n - 8);
      final x = liste[(idx + DateTime.now().weekday) % liste.length].split(':');
      _mot = x[0];
      _trad = x[1];
    }
  }

  void _touche(String k) {
    if (_fini || _mot.isEmpty) return;
    setState(() {
      if (k == '⌫') {
        if (_saisie.isNotEmpty) {
          _saisie = _saisie.substring(0, _saisie.length - 1);
        }
      } else if (k == 'OK') {
        if (_saisie.length != _mot.length) return;
        _essais.add(_saisie);
        _saisie = '';
        if (_essais.last == _mot) {
          _fini = true;
          widget.a.dire(_mot.toLowerCase());
          final n = _essais.length;
          final e = n <= 3 ? 3 : (n <= 5 ? 2 : 1);
          Future.delayed(const Duration(milliseconds: 900),
              () => widget.a.fin(e, 4 + 5 * e));
        } else if (_essais.length >= 6) {
          _fini = true;
          Future.delayed(const Duration(milliseconds: 1500),
              () => widget.a.fin(0, 0));
        }
      } else if (_saisie.length < _mot.length) {
        _saisie += k;
      }
    });
  }

  Color _couleur(String essai, int k, _LxP p) {
    if (essai[k] == _mot[k]) return _lxVert;
    if (_mot.contains(essai[k])) return _lxJaune;
    return p.gris;
  }

  Color? _couleurTouche(String l, _LxP p) {
    Color? c;
    for (final e in _essais) {
      for (var k = 0; k < e.length; k++) {
        if (e[k] != l) continue;
        if (_mot[k] == l) return _lxVert;
        c = _mot.contains(l) ? _lxJaune : p.gris.withOpacity(0.5);
      }
    }
    return c;
  }

  @override
  Widget build(BuildContext context) {
    final p = _lxP(context);
    if (_mot.isEmpty) {
      return Center(
        child: Padding(
          padding: const EdgeInsets.all(30),
          child: Text(
              'Ce jeu existe en anglais, espagnol et portugais. Change de langue pour y jouer 🦊',
              textAlign: TextAlign.center,
              style: TextStyle(color: p.texte, fontSize: 16)),
        ),
      );
    }
    final aide = widget.a.niveau <= 3 || _essais.length >= 3;
    const rangs = ['QWERTYUIOP', 'ASDFGHJKL', 'ZXCVBNM'];
    final cote = math.min(52.0, 300 / _mot.length);
    return ListView(
      padding: const EdgeInsets.fromLTRB(14, 6, 14, 16),
      children: [
        Text(
            aide
                ? 'Indice : 🇫🇷 ' + _trad
                : 'Trouve le mot de ' + _mot.length.toString() + ' lettres',
            textAlign: TextAlign.center,
            style: TextStyle(
                color: aide ? _lxOrange : p.gris,
                fontSize: 15,
                fontWeight: FontWeight.w700)),
        const SizedBox(height: 10),
        for (var r = 0; r < 6; r++)
          Padding(
            padding: const EdgeInsets.only(bottom: 6),
            child: Row(mainAxisAlignment: MainAxisAlignment.center, children: [
              for (var k = 0; k < _mot.length; k++)
                Builder(builder: (_) {
                  String l = '';
                  Color fond = p.carte;
                  Color bord = p.bord;
                  if (r < _essais.length) {
                    l = _essais[r][k];
                    fond = _couleur(_essais[r], k, p);
                    bord = fond;
                  } else if (r == _essais.length && k < _saisie.length) {
                    l = _saisie[k];
                    bord = widget.a.couleur;
                  }
                  return Container(
                    width: cote,
                    height: cote,
                    margin: const EdgeInsets.symmetric(horizontal: 3),
                    alignment: Alignment.center,
                    decoration: BoxDecoration(
                      color: fond,
                      borderRadius: BorderRadius.circular(8),
                      border: Border.all(color: bord, width: 2),
                    ),
                    child: Text(l,
                        style: TextStyle(
                            color: r < _essais.length ? Colors.white : p.texte,
                            fontSize: 22,
                            fontWeight: FontWeight.w900)),
                  );
                }),
            ]),
          ),
        if (_fini && _essais.last != _mot)
          Padding(
            padding: const EdgeInsets.only(top: 6),
            child: Text('Le mot était : ' + _mot + ' (' + _trad + ')',
                textAlign: TextAlign.center,
                style: const TextStyle(
                    color: _lxRouge, fontSize: 16, fontWeight: FontWeight.w800)),
          ),
        const SizedBox(height: 12),
        for (var i = 0; i < 3; i++)
          Padding(
            padding: const EdgeInsets.only(bottom: 6),
            child: Row(mainAxisAlignment: MainAxisAlignment.center, children: [
              if (i == 2) _clavier(p, 'OK', 1.6),
              for (final l in rangs[i].split('')) _clavier(p, l, 1),
              if (i == 2) _clavier(p, '⌫', 1.6),
            ]),
          ),
      ],
    );
  }

  Widget _clavier(_LxP p, String l, double f) {
    final c = l.length == 1 && l != '⌫' ? _couleurTouche(l, p) : null;
    return GestureDetector(
      onTap: () => _touche(l),
      child: Container(
        width: 30 * f,
        height: 44,
        margin: const EdgeInsets.symmetric(horizontal: 2),
        alignment: Alignment.center,
        decoration: BoxDecoration(
          color: c ?? (l == 'OK' ? _lxOrange : p.doux),
          borderRadius: BorderRadius.circular(7),
        ),
        child: Text(l,
            style: TextStyle(
                color: c != null || l == 'OK' ? Colors.white : p.texte,
                fontSize: 15,
                fontWeight: FontWeight.w800)),
      ),
    );
  }
}

// 10. Devine l'image

class _JeuImage extends StatefulWidget {
  const _JeuImage({required this.a});
  final _LxArgs a;
  @override
  State<_JeuImage> createState() => _JeuImageState();
}

class _JeuImageState extends State<_JeuImage> {
  late int _total;
  int _i = 0;
  int _vies = 3;
  int _score = 0;
  List<List<String>> _liste = [];
  List<String> _choix = [];
  String? _clique;

  int get _col =>
      {'en': 1, 'es': 2, 'pt': 3, 'zh': 4}[widget.a.langue] ?? 1;

  @override
  void initState() {
    super.initState();
    _total = 8 + widget.a.niveau;
    _liste = _lxImages.map((e) => e.toList()).toList()..shuffle(_lxR);
    _preparer();
  }

  void _preparer() {
    final bon = _liste[_i % _liste.length][_col];
    final nb = widget.a.niveau >= 6 ? 5 : 3;
    final autres = _lxImages
        .map((e) => e[_col])
        .where((x) => x != bon)
        .toSet()
        .toList()
      ..shuffle(_lxR);
    _choix = [bon, ...autres.take(nb)]..shuffle(_lxR);
    _clique = null;
  }

  void _repondre(String c) {
    if (_clique != null) return;
    final img = _liste[_i % _liste.length];
    final bon = c == img[_col];
    setState(() => _clique = c);
    widget.a.dire(img[_col]);
    if (bon) {
      _score += 2;
    } else {
      _vies--;
    }
    Future.delayed(Duration(milliseconds: bon ? 700 : 1300), () {
      if (!mounted) return;
      if (_vies <= 0) {
        widget.a.fin(0, _score);
        return;
      }
      if (_i + 1 >= _total) {
        widget.a.fin(_vies, _score + 5 * _vies);
        return;
      }
      setState(() {
        _i++;
        _preparer();
      });
    });
  }

  @override
  Widget build(BuildContext context) {
    final p = _lxP(context);
    final img = _liste[_i % _liste.length];
    return ListView(
      padding: const EdgeInsets.fromLTRB(18, 10, 18, 20),
      children: [
        Row(children: [
          Text((_i + 1).toString() + '/' + _total.toString(),
              style: TextStyle(
                  color: p.texte, fontSize: 15, fontWeight: FontWeight.w800)),
          const Spacer(),
          _lxVies(_vies, 3),
        ]),
        const SizedBox(height: 16),
        Center(
          child: Container(
            width: 170,
            height: 170,
            alignment: Alignment.center,
            decoration: BoxDecoration(
              color: p.carte,
              borderRadius: BorderRadius.circular(32),
              border: Border.all(color: p.bord, width: 2),
            ),
            child: Text(img[0], style: const TextStyle(fontSize: 96)),
          ),
        ),
        if (_clique != null)
          Padding(
            padding: const EdgeInsets.only(top: 10),
            child: Text('🇫🇷 ' + img[5],
                textAlign: TextAlign.center,
                style: TextStyle(color: p.gris, fontSize: 15)),
          ),
        const SizedBox(height: 20),
        if (_choix.length > 4)
          GridView.count(
            crossAxisCount: 2,
            shrinkWrap: true,
            physics: const NeverScrollableScrollPhysics(),
            crossAxisSpacing: 10,
            childAspectRatio: 2.6,
            children: [
              for (final c in _choix)
                _lxChoix(
                    p,
                    c,
                    _clique == null
                        ? null
                        : (c == img[_col]
                            ? _lxVert
                            : (c == _clique ? _lxRouge : null)),
                    () => _repondre(c)),
            ],
          )
        else
          for (final c in _choix)
            _lxChoix(
                p,
                c,
                _clique == null
                    ? null
                    : (c == img[_col]
                        ? _lxVert
                        : (c == _clique ? _lxRouge : null)),
                () => _repondre(c)),
      ],
    );
  }
}
